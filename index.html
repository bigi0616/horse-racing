async function loadRaceList(){ const key=document.getElementById('apiKey').value.trim(); if(!key){alert('인증키를 먼저 입력하세요.');return;} const date=document.getElementById('rdate').value.replace(/-/g,''); const meet=document.getElementById('meet').value; const container=document.getElementById('rcBtns'); container.innerHTML='경주 목록 확인 중...'; // 1~12경주 동시에 병렬 호출 const promises=Array.from({length:12},(_,i)=>i+1).map(async i=>{ try{ const res=await fetch(`/api/race?serviceKey=${encodeURIComponent(key)}&meet=${meet}&rc_date=${date}&rc_no=${i}`); const data=await res.json(); const items=data?.response?.body?.items?.item; return items?(Array.isArray(items)?items.length>0:true)?i:null:null; }catch(e){return null;} }); const results=await Promise.all(promises); const validRcNos=results.filter(n=>n!==null); if(validRcNos.length===0){ container.innerHTML='해당 날짜 경주 없음 (경마 개최일 토·일 확인)'; return; } container.innerHTML=validRcNos.map(no=> `` ).join(''); }
